# Python 3.12 or newer is REQUIRED:
#   numpy 2.5 and scipy (via python-doctr) need >= 3.12; torch 2.14, python-doctr 1.1,
#   Pillow 13 and pandas 3 need >= 3.11.
#
#   python -m venv .venv
#   .venv\Scripts\activate
#   python -m pip install --upgrade pip
#   pip install -r requirements.txt            (build tools: pip install -r requirements-dev.txt)
#
# GPU (Windows): the default PyPI torch wheel for Windows is CPU-only. Install the CUDA build FIRST:
#   pip install -r requirements-torch-gpu.txt   (CUDA 13.4 build from PyTorch's index; see that file for other tags)
#   pip install -r requirements.txt             (torch/torchvision below are ranges, so the CUDA build is kept)
# or in one go:  deploy_pyvenv.bat gpu
#
# See docs/modernization.md for why these versions were chosen.
# ---------------------------------------------------------------------

# --- OCR engine (PyTorch only since doctr 1.0; there is no "[torch]" extra any more) ---
# Ranges, not exact pins, on purpose: an exact pin (torch==2.14.1) is NOT satisfied by a CUDA build such as
# 2.15.0+cu134, so pip would replace it with the CPU wheel. Tested with torch 2.14.1+cu130 (Linux, CPU only)
# and torch 2.15.0 (CPU and +cu134 on a GTX 1660 Ti).
torch>=2.14.1,<3
torchvision>=0.29.2,<1
python-doctr==1.1.0

# --- Imaging / numerics ---

pypdf==7.1.0              # merging and overlaying text on existing PDF pages
pypdfium2==5.14.0         # core/pdf_input.py: rasterises PDF pages (no Ghostscript / Poppler needed)

# --- System / hardware monitoring ---
psutil==7.2.2
